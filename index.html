const SHEET_ID = '1O95DUE3dgFdgTUUbNk-etEaHTeFjbR7-tYVUCWkacYQ'; function doPost(e) { try { const datos = JSON.parse(e.postData.contents); if (datos.accion === 'guardarFactura' && datos.factura) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName('Facturas'); sheet.appendRow([ datos.factura.id || '', datos.factura.pacienteId || '', datos.factura.odontologoId || '', datos.factura.tratamientoId || '', datos.factura.total || 0, datos.factura.pagado || 0, datos.factura.deuda || 0, datos.factura.fecha || '' ]); Logger.log('Factura guardada: ' + datos.factura.id); } return ContentService.createTextOutput(JSON.stringify({ok: true})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { Logger.log('Error: ' + err.toString()); return ContentService.createTextOutput(JSON.stringify({error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
